# Read-only view of the current configuration.
# Reads the player's eotr_* character flags and renders a four-line summary
# of which option (if any) is selected for each setting. Confirming closes
# the panel with no side effects.
eotr_view_settings = {
	picture = { reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_compromise.dds" }
	decision_group_type = eotr_realm_directives

	desc = {
		desc = eotr_view_settings_intro

		desc = eotr_view_religion_label
		first_valid = {
			triggered_desc = { trigger = { has_character_flag = eotr_religion_convert } desc = eotr_view_religion_convert }
			triggered_desc = { trigger = { always = yes } desc = eotr_view_religion_none }
		}

		desc = eotr_view_culture_label
		first_valid = {
			triggered_desc = { trigger = { has_character_flag = eotr_culture_convert } desc = eotr_view_culture_convert }
			triggered_desc = { trigger = { has_character_flag = eotr_culture_accept } desc = eotr_view_culture_accept }
			triggered_desc = { trigger = { always = yes } desc = eotr_view_culture_none }
		}

		desc = eotr_view_priority_label
		first_valid = {
			triggered_desc = { trigger = { has_character_flag = eotr_priority_religion } desc = eotr_view_priority_religion }
			triggered_desc = { trigger = { has_character_flag = eotr_priority_culture } desc = eotr_view_priority_culture }
			triggered_desc = { trigger = { always = yes } desc = eotr_view_priority_none }
		}

		desc = eotr_view_default_label
		first_valid = {
			triggered_desc = { trigger = { has_character_flag = eotr_default_prosperity } desc = eotr_view_default_prosperity }
			triggered_desc = { trigger = { has_character_flag = eotr_default_bulwark } desc = eotr_view_default_bulwark }
			triggered_desc = { trigger = { has_character_flag = eotr_default_martial } desc = eotr_view_default_martial }
			triggered_desc = { trigger = { has_character_flag = eotr_default_none } desc = eotr_view_default_none }
			triggered_desc = { trigger = { always = yes } desc = eotr_view_default_unset }
		}
	}

	selection_tooltip = eotr_view_settings_tooltip
	confirm_text = eotr_view_settings_confirm

	is_shown = {
		is_landed = yes
		is_ai = no
	}

	is_valid_showing_failures_only = {}

	effect = {
		custom_tooltip = eotr_view_settings_effect_tooltip
	}

	ai_will_do = { base = 0 }
	ai_check_interval = 0
}

# Setting 1 of 4: Religion Policy
# Toggles the eotr_religion_convert character flag on the player.
eotr_set_religion_policy = {
	picture = { reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_compromise.dds" }
	decision_group_type = eotr_realm_directives

	desc = eotr_set_religion_policy_desc
	selection_tooltip = eotr_set_religion_policy_tooltip
	confirm_text = eotr_set_religion_policy_confirm

	is_shown = {
		is_landed = yes
		is_ai = no
	}

	is_valid_showing_failures_only = {}

	widget = {
		gui = "decision_view_widget_decision_option_list_controller"
		controller = decision_option_list_controller

		item = {
			value = eotr_religion_policy_convert
			current_description = eotr_religion_policy_convert_tooltip
			localization = eotr_religion_policy_convert_label
		}
		item = {
			value = eotr_religion_policy_none
			current_description = eotr_religion_policy_none_tooltip
			localization = eotr_religion_policy_none_label
		}
	}

	effect = {
		# Clear any prior selection in this category
		remove_character_flag = eotr_religion_convert

		# Apply the chosen option. Must use `if + limit = { scope:X = yes }`
		# rather than `scope:X ?= { ... }`: the latter shifts scope into the
		# option-list boolean tag, which is not a character, so add_character_flag
		# silently fails to persist.
		if = {
			limit = { scope:eotr_religion_policy_convert = yes }
			custom_tooltip = eotr_religion_policy_convert_effect_tooltip
			add_character_flag = eotr_religion_convert
		}
		else_if = {
			limit = { scope:eotr_religion_policy_none = yes }
			custom_tooltip = eotr_religion_policy_none_effect_tooltip
			# Nothing to add — already cleared
		}
	}

	ai_will_do = { base = 0 }
	ai_check_interval = 0
}

# Action: Issue Directives to the Realm
# Iterates every sub-realm vassal (cascading through tiers) and invokes the
# evaluator on each. Hidden when the player has no matching vassals.
eotr_apply_directives = {
	picture = { reference = "gfx/interface/illustrations/decisions/fp2_decision_struggle_compromise.dds" }
	decision_group_type = eotr_realm_directives

	desc = eotr_apply_directives_desc
	selection_tooltip = eotr_apply_directives_tooltip
	confirm_text = eotr_apply_directives_confirm

	is_shown = {
		is_landed = yes
		is_ai = no
		any_vassal_or_below = {
			is_landed = yes
			is_ai = yes
			highest_held_title_tier >= tier_county
		}
	}

	is_valid_showing_failures_only = {}

	effect = {
		custom_tooltip = eotr_apply_directives_effect_tooltip
		every_vassal_or_below = {
			limit = {
				is_landed = yes
				is_ai = yes
				highest_held_title_tier >= tier_county
			}
			eotr_evaluate_and_assign_directive = yes
		}
	}

	ai_will_do = { base = 0 }
	ai_check_interval = 0
}
